cmake_minimum_required(VERSION 3.24)
project(FullTextScopeIsolationProbe NONE)
include("${GUARD}")
add_library(product INTERFACE)
add_library(presentation INTERFACE)
target_link_libraries(product INTERFACE presentation)
if(MODE STREQUAL "source")
  # Real target membership, including a transitive production dependency.
  set_property(TARGET presentation PROPERTY SOURCES full_text_dictionary_scope_test.cpp)
elseif(MODE STREQUAL "interface-source")
  target_sources(presentation INTERFACE full_text_dictionary_scope_test.cpp)
elseif(MODE STREQUAL "dependency")
  add_custom_target(full_text_dictionary_scope_test)
  add_dependencies(presentation full_text_dictionary_scope_test)
elseif(MODE STREQUAL "genex")
  add_library(full_text_dictionary_scope_test INTERFACE)
  target_link_libraries(presentation INTERFACE "$<$<BOOL:1>:full_text_dictionary_scope_test>")
elseif(NOT MODE STREQUAL "clean")
  message(FATAL_ERROR "Unknown probe mode")
endif()
gd_check_full_text_scope_isolation(product)
